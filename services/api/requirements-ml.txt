# Heavy ML dependencies for OCR. These are installed into the linux/arm64
# Docker image (see services/api/Dockerfile + the repo-root docker-compose.yml),
# NOT natively on the host, and are kept separate from requirements.txt so
# `pnpm test:api` (which mocks the engine) stays fast and needs none of these
# large wheels.
#
# Why containerized: the macOS-arm64 paddlepaddle 2.6.2 CPU wheel hangs at 100%
# CPU on the first kernel dispatch (validated 2026-07-06), so a native install
# is unsupported on Apple Silicon. The linux/arm64 container dodges that wheel
# and runs paddle natively-fast on the same Apple Silicon hardware via Colima.
# The FIRST image build downloads and bakes the detection / recognition /
# angle-classification models into the image (one-time; needs network at build).
# OCR itself sends nothing to any external API.
#
# Pinned to the PaddleOCR 2.x API (use_angle_cls / use_gpu / .ocr(..., cls=)).
# Do NOT bump to 3.x — it renames .ocr() to .predict() and drops those kwargs.
paddleocr>=2.7,<2.10
paddlepaddle>=2.5,<2.7
numpy<2
opencv-python-headless>=4.6,<5
# Required: paddleocr's deps otherwise pull cython 3.2.x, which makes
# `import paddle` segfault in the Linux build. Validated in the spike.
cython<3.1
